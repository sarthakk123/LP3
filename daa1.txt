#include <iostream>
using namespace std;

// Recursive version
long long fibRecursive(int n) {
    if (n <= 1) return n;                       // base cases: F(0)=0, F(1)=1
    return fibRecursive(n - 1) + fibRecursive(n - 2);
}

// Non-recursive (iterative) version
long long fibIterative(int n) {
    if (n <= 1) return n;
    long long a = 0, b = 1;                     // F(0), F(1)
    for (int i = 2; i <= n; i++) {
        long long c = a + b;                    // next number
        a = b;
        b = c;
    }
    return b;
}

int main() {
    int n;
    cout << "Enter n: ";
    cin >> n;

    cout << "Recursive  F(" << n << ") = " << fibRecursive(n) << endl;
    cout << "Iterative  F(" << n << ") = " << fibIterative(n) << endl;
    return 0;
}
